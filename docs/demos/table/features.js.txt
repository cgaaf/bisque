// table-features.js: only sorting and pagination end up in the bundle.
import { createPaginatedRowModel, createSortedRowModel, rowPaginationFeature, rowSortingFeature, sortFns, tableFeatures } from "@tanstack/table-core";

export function orderTableFeatures() {
  return tableFeatures({
    rowSortingFeature,
    sortedRowModel: createSortedRowModel(),
    sortFns,
    rowPaginationFeature,
    paginatedRowModel: createPaginatedRowModel(),
  });
}
