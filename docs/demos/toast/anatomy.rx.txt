import { ToastProvider, ToastViewport, ToastRoot, ToastContent, ToastTitle, ToastDescription, ToastAction, ToastClose, toast } from "./bisque/toast/toast.rx"

component Toasts() {
  c = use toast
  for t in c.toasts {
    <ToastRoot toast={t}>
      <ToastContent>
        <ToastTitle />
        <ToastDescription />
        <ToastAction />
        <ToastClose />
      </ToastContent>
    </ToastRoot>
  }
}

<ToastProvider>
  <ToastViewport>
    <Toasts />
  </ToastViewport>
</ToastProvider>
