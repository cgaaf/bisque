import { Form } from "./rs-ui/form/form.rx"
import { FieldRoot, FieldLabel, FieldControl, FieldDescription, FieldError } from "./rs-ui/field/field.rx"

<Form onFormSubmit={(values) => { subscribe(values.email) }}>
  <FieldRoot name="email" validate={(value) => taken(value) ? "That address is already subscribed." : null}>
    <FieldLabel>Email</FieldLabel>
    <FieldControl type="email" required />
    <FieldDescription>We only write about your orders.</FieldDescription>
    <FieldError />
  </FieldRoot>
  <button type="submit">Subscribe</button>
</Form>
