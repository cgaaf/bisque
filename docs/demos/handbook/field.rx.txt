import { Form } from "./bisque/form/form.rx"
import { FieldRoot, FieldLabel, FieldControl, FieldDescription, FieldError } from "./bisque/field/field.rx"

// subscribe returns the request's promise: the form waits for it, and shows the errors it resolves to.
<Form onFormSubmit={(values) => subscribe(values.email)}>
  <FieldRoot name="email" validate={(value) => `${value}`.endsWith(".invalid") ? "Use a real address." : null}>
    <FieldLabel>Email</FieldLabel>
    <FieldControl type="email" required />
    <FieldDescription>We only write about your orders.</FieldDescription>
    <FieldError />
  </FieldRoot>
  <button type="submit">Subscribe</button>
</Form>
