import { TabsRoot, TabsList, TabsPanel, tabs, tabsListOptions, tabsTab } from "./bisque/tabs/tabs.rx"

component LinkTab(props: { value: string; href: string; children?: View }) {
  c = use tabs
  l = use tabsListOptions
  // The props function carries the whole part: roles, keys, selection, and registering with the root.
  t = tabsTab(c, l, props.value, false)
  <a href={props.href} {...t} type={undefined}>{props.children}</a>
}

<TabsRoot>
  <TabsList>
    <LinkTab value="inbox" href="#inbox">Inbox</LinkTab>
    <LinkTab value="sent" href="#sent">Sent</LinkTab>
  </TabsList>
  <TabsPanel value="inbox" />
  <TabsPanel value="sent" />
</TabsRoot>
